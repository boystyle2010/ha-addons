#include <tunables/global>

profile cloudflared_pro flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  capability,
  file,
  signal (send) set=(kill,term,int,hup,cont),

  # S6-Overlay, Bashio & Script permissions
  /init ix,
  /run.sh rix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/lib/bashio/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /etc/s6-overlay/** rix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Cloudflared binary execution
  /usr/bin/cloudflared cx -> cloudflared,

  profile cloudflared flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>
    #include <abstractions/ssl_certs>

    network inet dgram,
    network inet6 dgram,
    network inet stream,
    network inet6 stream,

    /usr/bin/cloudflared mr,
    /share/** rw,
    /ssl/** r,
    /data/** rw,
    /tmp/** rw,
    /proc/** r,
    /sys/** r,
  }
}
